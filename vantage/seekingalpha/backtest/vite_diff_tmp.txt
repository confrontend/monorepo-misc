+  const status = currentResearchStatus();
+  if (status.current) {

+  }

+    status: 'running',
+    startedAt: new Date().toISOString(),
+    finishedAt: null,
+    error: null,
+    log: '',
+  };
+  researchJob = job;
+
+  const researchDir = path.join(projectRoot, 'research');
+  const finish = (status: ResearchJob['status'], error: string | null) => {
+    job.status = status;
+    job.error = error;
+    job.finishedAt = new Date().toISOString();
+  };
+
+  // Prefer the workspace virtual environment so the UI uses the same pandas/numpy runtime as
+  // development. On a Windows-hosted Vite server that environment is a WSL venv, so launch it
+  // through wsl.exe instead of falling into the Microsoft Store `python` app-execution alias.
+  type PythonLaunch = { command: string; args: string[]; label: string };
+  // Unbuffered output lets the UI receive stage updates while Python is still running.
+  const analysisArgs = ['-u', 'run_analysis.py', '--out', 'report/'];
+  const workspaceRoot = path.resolve(projectRoot, '..', '..');
+  const windowsPython = path.join(workspaceRoot, '.venv', 'Scripts', 'python.exe');
+  const unixPython = path.join(workspaceRoot, '.venv', 'bin', 'python');
+  const toWslPath = (value: string) => {
+    const match = /^([A-Za-z]):[\\/](.*)$/.exec(value);
+    return match ? `/mnt/${match[1].toLowerCase()}/${match[2].replaceAll('\\', '/')}` : value.replaceAll('\\', '/');
+  };
+  const launches: PythonLaunch[] = process.platform === 'win32'
+    ? [
+        ...(existsSync(windowsPython) ? [{ command: windowsPython, args: analysisArgs, label: 'project Windows virtual environment' }] : []),
+        // Windows cannot reliably stat a Linux venv's `bin/python` symlink even though WSL can
+        // execute it, so this candidate must not be hidden behind existsSync().
+        ...([{
+          command: 'wsl.exe',
+          args: ['--cd', toWslPath(researchDir), '-e', toWslPath(unixPython), ...analysisArgs],
+          label: 'project WSL virtual environment',
+        }]),
+        { command: 'py', args: ['-3', ...analysisArgs], label: 'Windows Python launcher' },
+        { command: 'python', args: analysisArgs, label: 'system Python' },
+      ]
+    : [
+        ...(existsSync(unixPython) ? [{ command: unixPython, args: analysisArgs, label: 'project virtual environment' }] : []),
+        { command: 'python3', args: analysisArgs, label: 'system python3' },
+        { command: 'python', args: analysisArgs, label: 'system python' },
+      ];
+
+  const run = (index: number) => {
+    const launch = launches[index];
+    if (!launch) {
+      finish('failed', 'Python was not found. Expected the workspace virtual environment at .venv, or a working Python 3 installation.');
+      return;
+    }
+    job.log += `Starting analysis with ${launch.label}.\n`;
+    let retrying = false;
+    const child = spawn(launch.command, launch.args, { cwd: researchDir });
+    child.stdout.on('data', (chunk) => { job.log += String(chunk); });
+    child.stderr.on('data', (chunk) => { job.log += String(chunk); });
+    child.on('error', (error: NodeJS.ErrnoException) => {
+      if (error.code === 'ENOENT' && index + 1 < launches.length) {
+        retrying = true;
+        run(index + 1);
+        return;
+      }
+      finish('failed', error.code === 'ENOENT'
+        ? 'Python was not found. Expected the workspace virtual environment at .venv, or a working Python 3 installation.'
+        : error.message);
+    });
+    child.on('close', (code) => {
+      if (retrying || job.status !== 'running') return; // an error handler already retried/finished this job
+      if (code === 0) {
+        writeFileSync(researchDataVersionPath, String(currentResearchStatus().currentDataVersion), 'utf8');
+        finish('completed', null);
+        return;
+      }
+      if ((code === 9009 || code === 127) && index + 1 < launches.length) { run(index + 1); return; }
+      const missingModule = /ModuleNotFoundError: No module named '(\w+)'/.exec(job.log);
+      finish('failed', missingModule
+        ? `Python is installed but "${missingModule[1]}" is missing. Install it with: pip install pandas numpy`
+        : `Analysis exited with code ${code}. ${job.log.trim().split('\n').slice(-3).join(' ')}`);
+    });
+  };
+  run(0);
+  return job;
+};
